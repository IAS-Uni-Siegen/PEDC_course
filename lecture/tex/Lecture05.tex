\begin{frame}{Introduction to Power MOSFETs}
    \begin{columns}
        \begin{column}{0.6\textwidth}
\begin{itemize}
    \item Power MOSFETs (Metal-Oxide-Semiconductor Field Effect Transistors) are widely used in power electronics since the early 1980s.
    \item They offer high on-state current capability and off-state voltage blocking, making them suitable for high-speed switching applications.
    \item Replacing BJTs in many applications, especially those requiring fast switching, due to their voltage-controlled nature.
    \item We learn about the physical structure, voltage and current limits, and failure mechanisms of MOSFETs.
\end{itemize}

\begin{frame}{Basic Structure of Power MOSFET}
                \begin{columns}
        \begin{column}{0.48\textwidth}
\begin{itemize}
    \item The MOSFET is constructed with alternating p-type and n-type layers: \textbf{n$^+$-p-n$^-$-n$^+$}.
    \item The n$^+$ (source and drain), p-body, and n$^-$ (drift region) form the core active structure.
    \item Gate terminal is insulated from the body by silicon dioxide (SiO$_2$) forming the gate oxide.
    \item Applying a positive voltage to the gate induces an n-channel in the p-body, allowing current flow.
    \item The drift region (n$^-$) is lightly doped to handle high blocking voltages.
\end{itemize}
     \end{column}

\begin{frame}{Perspective View of a Power MOSFET}
                    \begin{columns}
        \begin{column}{0.5\textwidth}
\begin{itemize}
    \item Power MOSFETs consist of thousands of small identical cells connected in parallel.
    \item Each cell includes source diffusion, gate conductor, field oxide, and drain contact.
    \item High current handling is achieved by maximizing the number of cells and gate width.
    \item This configuration is often referred to as VDMOS (Vertical Diffused MOSFET).
    \item Parasitic elements like a BJT and integral body diode are naturally formed.
\end{itemize}

\begin{frame}{Channel Formation and Operation}
                    \begin{columns}
        \begin{column}{0.5\textwidth}
\begin{itemize}
    \item The p-body prevents current flow until a gate voltage is applied.
    \item Applying V$_{GS} > V_{th}$ induces an n-channel in the p-body.
    \item This connects the source to the drain allowing electron flow (n-channel enhancement).
    \item No minority carrier injection occurs, enabling fast switching.
    \item The MOSFET operates as a unidirectional voltage-controlled switch.
\end{itemize}

\begin{frame}{Gate Metallization and Field Plate Action}
                    \begin{columns}
        \begin{column}{0.5\textwidth}
\begin{itemize}
    \item Gate metallization overlaps the n$^-$ drift region to serve two purposes:
    \begin{enumerate}
        \item Enhances conductivity during ON-state by forming accumulation layer.
        \item Acts as a field plate to reduce electric field curvature in OFF-state.
    \end{enumerate}
    \item This helps to minimize on-resistance and improves breakdown voltage.
    \item Design strategies include optimizing gate width-to-length ratio to maximize gain.
\end{itemize}

\begin{frame}{Inversion Layers and the Field Effect}
\begin{itemize}
    \item The gate region of the MOSFET is composed of:
    \begin{itemize}
        \item Gate metallization (e.g., Al or polysilicon)
        \item Gate oxide ($SiO_2$)
        \item Silicon substrate beneath the oxide
    \end{itemize}
    \item This gate stack acts as a high-quality MOS capacitor.
    \item Applying a small positive $V_{\mathrm{GS}}$ causes:
    \begin{itemize}
        \item Positive charge on gate metal
        \item Negative charge induced on silicon side
        \item Formation of a depletion region by repelling holes
    \end{itemize}
    \item The structure acts as a capacitor inducing a field across the gate oxide.
\end{itemize}
\vspace{0.5em}
\textbf{Note:} The inversion layer forms only after depletion is sufficient.

\end{frame}


\begin{frame}{Formation of Depletion and Inversion Layers}
\begin{columns}
\column{0.5\textwidth}
\begin{itemize}
    \item As $V_{\mathrm{GS}}$ increases:
    \begin{enumerate}
        \item Depletion region widens (\ref{fig:MOSFET_op} (a))
        \item Electrons begin to accumulate at oxide–silicon interface (\ref{fig:MOSFET_op} (b))
        \item Inversion layer forms when electron density exceeds hole density (\ref{fig:MOSFET_op} (c))
    \end{enumerate}
    \item Inversion layer has \textbf{n-type} conductivity and enables conduction between source and drain.
    \item This defines the MOSFET’s enhancement mode behavior.
\end{itemize}

\begin{frame}{Threshold Voltage and Oxide Capacitance}
\begin{itemize}
    \item Threshold voltage $V_{\mathrm{GS(th)}}$ is the point at which an inversion layer is just formed.
    \item As $V_{\mathrm{GS}} > V_{\mathrm{GS(th)}}$, inversion layer:
    \begin{itemize}
        \item Becomes thicker
        \item Becomes more conductive
        \item Screens the underlying depletion region
    \end{itemize}
    \item Major factor: oxide capacitance per unit area:
    \[
        C_{\mathrm{ox}} = \frac{\varepsilon_{\mathrm{ox}}}{t_{\mathrm{ox}}}
    \]
    \item Other influencing factors:
    \begin{itemize}
        \item Work function difference (metal vs silicon)
        \item Fixed/trapped charge
        \item Body doping and oxide thickness
    \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Gate Control of Drain Current Flow}
\begin{itemize}
    \item With $V_{\mathrm{GS}} > V_{\mathrm{GS(th)}}$ and small $V_{\mathrm{DS}}$:
    \begin{itemize}
        \item MOSFET operates in the ohmic (linear) region.
        \item Inversion layer has nearly uniform thickness along the channel.
        \item Drain current $I_D$ increases linearly with $V_{\mathrm{DS}}$.
    \end{itemize}
    \item Voltage drop along the channel varies:
    \[
        V_{\mathrm{ox}}(x) = V_{\mathrm{GS}} - V_{\mathrm{CS}}(x)
    \]
    \item As $V_{\mathrm{DS}}$ increases, $V_{\mathrm{CS}}(x)$ increases toward the drain end.
    \item This leads to non-uniform channel thickness and begins shaping current characteristics.
\end{itemize}
\end{frame}


\begin{frame}{Channel Pinch-Off and Saturation Region}
\begin{itemize}
    \item As $V_{\mathrm{DS}}$ increases such that:
    \[
        V_{\mathrm{GS}} - V_{\mathrm{DS}} = V_{\mathrm{GS(th)}}
    \]
    the channel is \textbf{pinched off} at the drain end.
    \item No inversion layer exists at the drain end—carrier velocity becomes saturated.
    \item Device enters saturation (active) region:
    \begin{itemize}
        \item $I_D$ becomes independent of $V_{\mathrm{DS}}$
        \item Minimum inversion layer thickness is maintained by high electric field
    \end{itemize}
    \item Channel length modulation and velocity saturation effects begin.
\end{itemize}
\end{frame}


\begin{frame}{Electric Field and Velocity Saturation}
\begin{itemize}
    \item At high electric fields, carrier velocity saturates:
    \[
        v_{\mathrm{drift}} \approx 8 \times 10^6~\text{cm/s at } E \approx 1.5 \times 10^4~\text{V/cm}
    \]
    \item This saturation leads to constant $I_D$ in the saturation region:
    \[
        I_D = K (V_{\mathrm{GS}} - V_{\mathrm{GS(th)}})^2
    \]
    with 
    \[
        K = \mu_n C_{\mathrm{ox}} \frac{W}{2L}
    \]
    \item $\mu_n$ is carrier mobility; $W/L$ is width-to-length ratio of the channel.
    \item Design goal: maximize $W/L$ to minimize on-state losses and improve gain.
\end{itemize}
\end{frame}


\begin{frame}{Deviation from Square Law Model}
\begin{itemize}
    \item Square-law $I_D$–$V_{\mathrm{GS}}$ relationship:
    \[
        I_D \propto (V_{\mathrm{GS}} - V_{\mathrm{GS(th)}})^2
    \]
    \item Breaks down at high $I_D$ due to:
    \begin{itemize}
        \item Velocity saturation (as per $v_{\mathrm{drift}}$ curve)
        \item Reduction in mobility $\mu_n$ due to:
        \begin{itemize}
            \item Increased electric field in inversion layer
            \item Increased electron density near gate
            \item Carrier–carrier scattering effects
        \end{itemize}
    \end{itemize}
    \item At high currents, the relationship becomes approximately linear.
\end{itemize}
\end{frame}

    \[

    \begin{itemize}
        \item $V_{GS} < V_{GS(th)}$
        \item No conductive channel exists
        \item MOSFET behaves like an open switch

\end{itemize}

\begin{frame}{MOSFET Transfer Characteristics and Current Equations}
\begin{columns}[T]
\column{0.58\textwidth}
\textbf{Drain Current in the Active Region}
\begin{itemize}
    \item In the active region, the drain current approximately follows:
    \[
    i_D = K(V_{GS} - V_{GS(th)})^2
    \]
    where:
    \begin{itemize}
        \item $K$ : device constant
        \item $V_{GS(th)}$ : threshold voltage
    \end{itemize}
    \item This equation is derived from first-order MOSFET theory.
\end{itemize}

\textbf{Boundary Between Regions}
\begin{itemize}
    \item The transition between ohmic and active regions occurs when:
    \[
    v_{GS} - V_{GS(th)} = v_{DS}
    \]
\end{itemize}

\begin{frame}{Dynamic Characteristics of Power MOSFETs}
\begin{columns}[T]
\column{0.55\textwidth}
\textbf{Why MOSFETs Switch Faster than BJTs}
\begin{itemize}
    \item MOSFETs are majority-carrier devices.
    \item No minority carrier storage exists during switching.
    \item Switching speed is mainly determined by charging and discharging:
    \begin{itemize}
        \item Gate-source capacitance $C_{gs}$
        \item Gate-drain capacitance $C_{gd}$
        \item Drain-source capacitance $C_{ds}$
    \end{itemize}
    \item Gate current is primarily capacitive current.
\end{itemize}

\textbf{Important Switching Waveforms}
\begin{itemize}
    \item Gate voltage $V_{gs}$
    \item Drain current $i_d$
    \item Drain-source voltage $V_{ds}$

\begin{frame}{Turn-ON Process of a MOSFET}
\begin{columns}[T]
\column{0.56\textwidth}
\textbf{Interval $t_0 - t_1$}
\begin{itemize}
    \item Gate-source capacitance begins charging. 
    \item Gate voltage rises threshold voltage $V_{GS(th)}$.
    \item Drain current is still nearly zero.
\end{itemize}

\textbf{Interval $t_1 - t_2$}
\begin{itemize}
    \item MOSFET channel begins forming.
    \item Drain current rises tofull load current.
    \item Device starts entering conduction mode.
\end{itemize}

\textbf{Interval $t_2 - t_3$}
\begin{itemize}
    \item Gate-drain capacitance $C_{gd}$ dominates.
    \item Drain-source voltage falls rapidly.
    \item Gate voltage remains nearly constant:
    \[
    \text{Miller Plateau}
    \]
\end{itemize}

\textbf{Interval $t_3 - t_4$}
\begin{itemize}
    \item MOSFET reaches full enhancement.
    \item $R_{DS(on)}$ becomes minimum.
    \item Device fully enters ON-state.
\end{itemize}

\begin{frame}{Turn-OFF Process and Switching Equations}
\begin{columns}[T]
\column{0.58\textwidth}
\textbf{Turn-OFF Intervals}
\begin{itemize}
    \item \textbf{$t_5 - t_6$}
    \begin{itemize}
        \item Excess gate charge is removed.
        \item Device exits full enhancement.
    \end{itemize}
    
    \item \textbf{$t_6 - t_7$}
    \begin{itemize}
        \item Drain voltage rises.
        \item Feedback capacitance $C_{gd}$ discharges.
        \item MOSFET leaves ohmic region.
    \end{itemize}
    
    \item \textbf{$t_7 - t_8$}
    \begin{itemize}
        \item Gate-source capacitance fully discharges.
        \item Drain current falls to zero.
        \item Device reaches OFF-state.
    \end{itemize}
\end{itemize}

\column{0.42\textwidth}
\textbf{Important Dynamic Relations}
\[
i_{gs} = C_{gs}\frac{dV_{gs}}{dt}
\]

\[
i_{gd} = C_{gd}\frac{dV_{gd}}{dt}
\]

\[
\frac{dV_{ds}}{dt} = \frac{i_g}{C_{gd}}
\]

\end{columns}
\end{frame}

\begin{frame}{Paralleling and Power Dissipation of MOSFETs}
\begin{columns}[T]
\column{0.55\textwidth}
\begin{itemize}
    \item MOSFETs can be connected in parallel for higher current capability.
    \item Unlike BJTs, MOSFETs exhibit: positive temperature coefficient of $R_{DS(on)}$.
    \item If current increases in one device:
    \begin{itemize}
        \item Junction temperature rises
        \item $R_{DS(on)}$ increases
        \item Current naturally redistributes
    \end{itemize}
    \item This provides inherent current balancing.
\end{itemize}

\textbf{Gate Resistance Requirement}
\begin{itemize}
    \item Small gate resistors are used:  4$\Omega$ to 100 $\Omega$
    \item They help suppress: oscillations, ringing, and high-frequency instability.
\end{itemize}

\begin{frame}{Introduction to IGBT}
\begin{itemize}
    \item The insulated gate bipolar transistor (IGBT) combines useful features of both MOSFETs and BJTs.
    \item A MOSFET offers fast switching and voltage-controlled gate operation, but its conduction loss becomes high at larger blocking voltages.
    \item A BJT offers lower on-state conduction loss at high voltage ratings, but requires continuous base current and has slower turn-off due to stored minority carriers.
    \item The IGBT uses a MOS gate structure to control a bipolar conduction path.
    \item Therefore, it behaves as a voltage-controlled power switch with conductivity modulation in the drift region.
    \item It is widely used in medium- and high-power converters because it offers a good compromise between conduction loss, switching speed, and gate-drive simplicity.
\end{itemize}
\end{frame}


\begin{frame}{Basic Structure of an IGBT}
\begin{columns}
\begin{column}{0.5\textwidth}
\begin{itemize}
    \item A generic n-channel IGBT has a vertical layered structure: $n^+ - p - n^- - n^+ - p^+$
    \item The upper part resembles a vertical power MOSFET with source, gate, oxide, p-body, and n$^-$ drift region.
    \item Key difference from a MOSFET is the additional p$^+$ injecting layer at drain side.
    \item This p$^+$ layer forms the junction $J_1$ and injects holes into the n$^-$ drift region during conduction.
    \item The n$^-$ drift region supports the blocking voltage in the OFF-state.
    \item The body-source short suppresses unwanted turn-on of the parasitic thyristor.
\end{itemize}
\end{column}

\end{frame}


\begin{frame}{Punch-Through and Non-Punch-Through IGBTs}
\begin{itemize}
    \item The n$^-$ drift region is responsible for supporting the blocking voltage of the IGBT.
    \item In a non-punch-through (NPT) IGBT, the drift region is sufficiently thick so that the depletion layer does not reach the p$^+$ injecting layer.
    \item NPT structures can provide approximately symmetrical blocking capability, meaning the reverse-blocking voltage can be comparable to the forward-blocking voltage.
    \item In a punch-through (PT) IGBT, an n$^+$ buffer layer is inserted between the p$^+$ injecting layer and the n$^-$ drift region.
    \item The buffer layer prevents the depletion region from reaching the p$^+$ layer even when the drift region is made thinner.
    \item PT IGBTs have lower on-state voltage because of the shorter drift region, but their reverse-blocking capability is poor.
\end{itemize}

\[
\text{Shorter drift region} \Rightarrow \text{lower } V_{CE(on)}
\quad \text{but weaker reverse blocking}
\]

\end{frame}


\begin{frame}{IGBT Blocking-State Operation}
\begin{columns}
\begin{column}{0.5\textwidth}
\begin{itemize}
    \item The gate-source voltage controls the IGBT in a manner similar to a MOSFET.
    \item When
    \[
    v_{GS} < V_{GS(th)},
    \]
    no inversion channel is formed below the gate oxide.
    \item The source is therefore not electrically connected to the n$^-$ drift region through a MOS channel.
    \item In this condition, the IGBT is in the OFF-state and supports the applied drain-source or collector-emitter voltage.
\end{itemize}
\end{column}

\begin{column}{0.48\textwidth}
\begin{itemize}
    \item Most of the applied voltage is dropped across the reverse-biased junction $J_2$.
    \item The depletion region extends mainly into the lightly doped n$^-$ drift region, while only a small leakage current flows.
\end{itemize}
\begin{block}{Physical Meaning}
The blocking capability is governed mainly by the doping and thickness of the n$^-$ drift region. A lightly doped and sufficiently thick drift region is required for high voltage blocking.
\end{block}
\end{column}
\end{columns}
\end{frame}


\begin{frame}{IGBT On-State Operation}
\begin{columns}
\begin{column}{0.5\textwidth}
\begin{itemize}
    \item When gate-source voltage exceeds threshold voltage, $v_{GS} > V_{GS(th)}$, an inversion channel forms in the p-body region.
    \item Electrons flow from n$^+$ source through inversion channel into n$^-$ drift region.
    \item This electron current enables hole injection from the p$^+$ drain/injecting layer into the n$^-$ drift region.
    \item The injected holes and electrons increase the carrier concentration in the drift region, called conductivity modulation.
    \item Conductivity modulation reduces the drift-region resistance and gives the IGBT lower on-state voltage than HV MOSFET.
\end{itemize}
\end{column}

\begin{frame}{Equivalent Circuit and On-State Voltage of IGBT}
\begin{columns}
\begin{column}{0.52\textwidth}
\begin{itemize}
    \item The IGBT can be interpreted as a MOSFET driving a bipolar transistor.
    \item The MOSFET channel controls the electron current from source to drift region.
    \item The p$^+$ injecting layer, n$^-$ drift region, and p-body form an effective pnp transistor.
    \item A simplified on-state voltage expression is:
    \[
    V_{DS(on)} = V_{J1} + V_{drift} + I_D R_{channel}
    \]
    \item $V_{J1}$ is the forward voltage drop of the injection junction.
    \item $V_{drift}$ is reduced by conductivity modulation, making IGBT attractive at high voltage ratings.
\end{itemize}
\end{column}

\begin{frame}{Latch-Up Mechanism in IGBTs}
\begin{itemize}
    \item The IGBT structure contains a parasitic thyristor formed by coupled pnp and npn transistor sections.
    \item During normal operation, the body-source short and cell design prevent this parasitic thyristor from turning on.
    \item Hole current injected into the n$^-$ drift region partly flows laterally through the p-body region toward the source contact.
    \item This lateral current produces a voltage drop across the body spreading resistance.
    \item If this voltage drop becomes large enough, the n$^+$ source--p-body junction $J_3$ becomes forward biased.
    \item Electron injection from the n$^+$ source can then activate the parasitic npn transistor.
    \item Once both parasitic npn and pnp transistors conduct, latch-up occurs and the gate loses control of the drain current.
\end{itemize}

$\text{Latch-up} \Rightarrow \text{gate control is lost} \Rightarrow \text{device can be destroyed by excessive current}$
\end{frame}


\begin{frame}{Avoidance of Latch-Up in IGBTs}
\begin{columns}
\begin{column}{0.5\textwidth}
\begin{itemize}
    \item Latch-up is avoided by preventing excessive lateral voltage drop in the p-body region.
    \item The source/body geometry is designed to reduce the body spreading resistance.
    \item The lateral width of the source region is kept small so that hole current does not travel a long distance beneath the source.
    \item Heavier p-body doping beneath the source reduces the local resistance and raises the latch-up current threshold.
    \item A hole-bypass structure can provide an alternate path for the hole current.
    \item Slower turn-off using suitable gate resistance can also reduce dynamic latch-up risk.
\end{itemize}
\end{column}

\begin{frame}{Key Takeaways: IGBT Device Physics}
\begin{itemize}
    \item The IGBT combines MOS gate control with bipolar conduction.
    \item The MOS channel controls electron flow, while the p$^+$ injecting layer enables hole injection into the drift region.
    \item Conductivity modulation lowers the drift-region resistance and reduces conduction loss compared with high-voltage MOSFETs.
    \item The n$^-$ drift region determines the blocking voltage and dominates the trade-off between voltage rating and on-state loss.
    \item PT structures use an n$^+$ buffer layer to reduce drift-region thickness and on-state voltage, but with poor reverse-blocking capability.
    \item The parasitic thyristor inside the IGBT can cause latch-up if the p-body lateral voltage drop becomes excessive.
    \item Modern IGBTs use optimized body-source geometry, body shorting, buffer layers, and hole-bypass concepts to minimize latch-up risk.
\end{itemize}
\end{frame}

\begin{frame}{Static Characteristics of an IGBT}
\begin{columns}
\begin{column}{0.6\textwidth}
\begin{itemize}
    \item The static characteristics of an n-channel IGBT describe the relation between collector current and collector--emitter voltage: $i_C = f(v_{CE},V_{GE})$.
    \item Similar to a MOSFET, the gate-emitter voltage $V_{GE}$ controls the current level.
    \item Similar to a BJT, the on-state current path includes bipolar carrier injection, which reduces the drift-region resistance.
    \item When $V_{GE}<V_{GE(th)}$, no inversion channel is formed and the IGBT remains in the OFF-state.
    \item When $V_{GE}$ increases above threshold, the collector current increases and the device enters forward conduction.
\end{itemize}
\end{column}

\begin{frame}{Output Characteristics: Forward Conduction}
\begin{columns}
\begin{column}{0.65\textwidth}
\begin{itemize}
    \item In the forward direction, the IGBT output characteristics resemble those of a BJT.
    \item However, the controlling variable is not base current, but the gate-emitter voltage $V_{GE}$.
    \item Increasing $V_{GE}$ increases the channel current and therefore increases the collector current.
    \item The forward characteristics show a knee voltage because conduction requires both:
    \begin{itemize}
        \item formation of the MOS inversion channel, and
        \item forward-biased carrier injection from the p$^+$ collector layer.
    \end{itemize}
    \item At high current levels, the IGBT has a relatively low on-state voltage because the n$^-$ drift region is conductivity modulated.
\end{itemize}
\end{column}

\begin{frame}{Forward and Reverse Blocking Characteristics}
\begin{itemize}
    \item In the OFF-state, the IGBT must support the applied collector--emitter voltage while allowing only a small leakage current.
    \item Forward blocking occurs when the collector is positive with respect to the emitter and $V_{GE}<V_{GE(th)}$.
    \item The junction $J_2$ blocks the applied forward voltage, and the depletion layer extends mainly into the lightly doped n$^-$ drift region.
    \item Reverse blocking depends strongly on the internal IGBT structure.
    \item In a non-punch-through IGBT without an n$^+$ buffer layer, the reverse-blocking voltage can be made comparable to the forward-blocking voltage.
    \item In a punch-through IGBT with an n$^+$ buffer layer, reverse-blocking capability is very poor because the reverse-blocking junction has heavily doped material on both sides.
\end{itemize}

\begin{block}{Key Insight}
$\text{NPT IGBT} \Rightarrow \text{possible reverse blocking}
\qquad
\text{PT IGBT} \Rightarrow \text{low reverse blocking}$
\end{block}
\end{frame}


\begin{frame}{Comparison with MOSFET and BJT Static Behavior}

\begin{itemize}
    \item The IGBT input characteristic is MOSFET-like because the gate is insulated and voltage-controlled.
    \item The IGBT output characteristic is BJT-like because the forward conduction path involves minority-carrier injection.
    \item For a given current, the on-state voltage of an IGBT is usually:
    \[
        V_{CE(sat),IGBT} < V_{DS(on),MOSFET}
    \]
    for comparable high-voltage ratings.
    \item However, compared with a BJT:
    \[
        V_{CE(sat),IGBT} > V_{CE(sat),BJT}
    \]
    because the IGBT still includes a MOS channel resistance and internal junction drops.
    \item Thus, the IGBT is especially attractive where high blocking voltage and moderate switching frequency are required.
\end{itemize}

\end{frame}


\begin{frame}{Comparison with MOSFET and BJT Static Behavior}
\begin{block}{Key point}
The IGBT should be understood as a hybrid power device:
\begin{itemize}
    \item \textbf{MOS gate} gives easy voltage control.
    \item \textbf{Bipolar conduction} gives low drift-region resistance.
    \item \textbf{Static characteristics} combine features of MOSFETs and BJTs.
\end{itemize}
\end{block}

\begin{block}{Practical Interpretation}
In power converters, the IGBT is usually operated either in:
\begin{itemize}
    \item OFF-state blocking region, or
    \item ON-state forward conduction region.
\end{itemize}
The transition region is mainly important for switching loss analysis.
\end{block}
\end{frame}

\begin{frame}{Turn-OFF Characteristics and Tail Current}
\begin{columns}
\begin{column}{0.52\textwidth}
\begin{itemize}
    \item During turn-OFF, the IGBT behaves more like a bipolar device.
    
    \item The turn-OFF sequence consists of:
    \begin{itemize}
        \item turn-OFF delay time $t_{d(off)}$,
        \item voltage rise time,
        \item current fall interval.
    \end{itemize}
    
    \item The tail current is caused by stored minority carriers in the n$^-$ drift region.
    
    \item These carriers cannot be removed instantly and must recombine within the device.
    
    \item The tail interval increases switching loss because: $v_{CE} \approx V_{DC}; \text{while};  i_C \neq 0$.    
    \item Therefore, a trade-off exists between:
    \begin{itemize}
        \item low on-state losses, and
        \item fast turn-OFF speed.
    \end{itemize}
\end{itemize}
\end{column}

\begin{frame}{Tail Current Mechanism in IGBTs}
\begin{columns}
\begin{column}{0.56\textwidth}
\begin{itemize}
    \item The tail current originates from excess holes stored in the lightly doped n$^-$ drift region.
    
    \item During conduction, conductivity modulation reduces the drift-region resistance and improves efficiency.
    
    \item However, during turn-OFF these carriers remain stored temporarily.
    
    \item Since no reverse voltage exists to sweep out the holes rapidly, carrier recombination becomes the dominant removal mechanism.
    
    \item The tail current duration depends strongly on:
    \begin{itemize}
        \item carrier lifetime,
        \item drift-region thickness,
        \item device structure (PT or NPT),
        \item operating temperature.
    \end{itemize}
    
    \item Higher temperature generally increases carrier lifetime and prolongs the tail interval.
\end{itemize}
\end{column}

\begin{column}{0.44\textwidth}
\begin{block}{Engineering Trade-Off}
$\text{Low } V_{CE(sat)} \Longleftrightarrow \text{Large stored charge}$
$\text{Large stored charge} \Longrightarrow \text{Longer tail current}$
\end{block}

\vspace{0.15cm}

\begin{block}{Design Objective}
Modern IGBT structures attempt to:
\begin{itemize}
    \item reduce tail current,
    \item maintain low conduction loss,
    \item improve switching speed.
\end{itemize}
\end{block}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{PT and NPT IGBT Structures}
\begin{columns}
\begin{column}{0.6\textwidth}
\begin{itemize}
    \item Two important IGBT structures are:
    \begin{itemize}
        \item Punch-Through (PT),
        \item Non-Punch-Through (NPT).
    \end{itemize}
    
    \item PT IGBTs use an n$^+$ buffer layer between the drift region and p$^+$ collector.
    
    \item The buffer layer helps remove stored carriers more rapidly and shortens the tail current interval.
    
    \item PT devices generally provide:
    \begin{itemize}
        \item lower turn-OFF losses,
        \item shorter switching times,
        \item improved high-frequency operation.
    \end{itemize}
    
    \item NPT IGBTs use thicker drift region, usually exhibit:
    \begin{itemize}
        \item better ruggedness,
        \item superior avalanche capability,
        \item improved high-voltage behavior.
    \end{itemize}
\end{itemize}
\end{column}

\begin{column}{0.4\textwidth}
\begin{block}{PT IGBT $\Rightarrow \text{speed}$}
\begin{itemize}
    \item Faster switching
    \item Smaller tail current
    \item Better for medium-voltage operation
\end{itemize}
\end{block}

\vspace{0.15cm}

\begin{block}{NPT IGBT $\Rightarrow \text{ruggedness}$}
\begin{itemize}
    \item Higher robustness
    \item Better thermal stability
    \item Preferred at higher voltages
\end{itemize}
\end{block}
\end{column}
\end{columns}
\end{frame}


\begin{frame}{Device Limits of IGBTs}
\begin{columns}
\begin{column}{0.5\textwidth}
\begin{itemize}
    \item The maximum collector current is limited to avoid:
    \begin{itemize}
        \item excessive thermal stress,
        \item bond-wire failure,
        \item latch-up conditions.
    \end{itemize}
    
    \item The maximum gate-emitter voltage is limited by the gate oxide strength:
    \[
        |V_{GE(max)}| \approx 20\text{ V}
    \]
    for many practical devices.
    
    \item The maximum collector-emitter voltage is determined by avalanche breakdown of the drift region.
    
\end{itemize}
\end{column}

\begin{block}{Important Device Limits}
\begin{itemize}
    \item $V_{GE(max)}$
    \item $V_{CES}$
    \item $I_C$
    \item $I_{CM}$
    \item $T_J$
    \item Short-circuit withstand time
\end{itemize}
\end{block}

\end{columns}

\begin{frame}{Device Limits of IGBTs}
\begin{columns}
\begin{column}{0.54\textwidth}
\begin{itemize}   
    \item IGBTs are designed to survive short-circuit conditions for a limited duration:
    \[
        t_{SC} \approx 5{-}10~\mu s
    \]
    
    \item Modern devices can operate at junction temperatures up to:
    \[
        T_J \approx 150^\circ\text{C}
    \]
    
    \item Positive temperature coefficient behavior helps current sharing in parallel-connected IGBTs.
\end{itemize}
\end{column}

IGBTs can tolerate much larger overload currents than MOSFETs for short durations because conductivity modulation distributes current more effectively within the drift region.
\end{block}
\end{column}
\end{columns}

\begin{frame}{Remarks on Dynamic Performance of IGBTs}
\begin{itemize}
    \item IGBTs combine the advantages of MOSFET gate control with the low conduction losses of bipolar devices.
    
    \item The gate-drive power requirement is relatively low because the gate is insulated.
    
    \item Compared with MOSFETs:
    \begin{itemize}
        \item IGBTs have lower conduction loss at high voltage,
        \item but higher switching losses due to stored charge.
    \end{itemize}
    
    \item Compared with BJTs:
    \begin{itemize}
        \item IGBTs do not require continuous base current drive,
        \item and exhibit easier gate control.
    \end{itemize}
    
    \item Turn-ON behavior is largely MOSFET-like, while turn-OFF behavior strongly reflects bipolar carrier dynamics.
    
    \item Proper gate-drive design is extremely important to:
    \begin{itemize}
        \item reduce switching losses,
        \item suppress oscillations,
        \item avoid excessive $dv/dt$ and $di/dt$,
        \item maintain SOA compliance.
    \end{itemize}
\end{itemize}
\end{frame}

\begin{block}{Design Relevance}
IGBT datasheet parameters help the designer evaluate:
\begin{itemize}
    \item \textbf{Voltage and current limits:} $V_{CE}$, $I_C$, $I_{C,pulse}$
    \item \textbf{Conduction loss:} mainly through $V_{CE(sat)}$
    \item \textbf{Switching loss:} mainly through $E_{on}$, $E_{off}$, $t_r$, and $t_f$
    \item \textbf{Gate-driver design:} $V_{GE(max)}$, $V_{GE(th)}$, $Q_G$
    \item \textbf{Thermal design:} $R_{\theta JC}$, $R_{\theta JA}$, $T_{j(max)}$
    \item \textbf{Reliability margin:} short-circuit rating, diode recovery, and SOA
\end{itemize}
\end{block}
